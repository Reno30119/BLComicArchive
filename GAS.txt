/**
 * 程式夥伴優化版：提升讀取速度並整理結構
 */
function doGet(e) {
  try {
    var action = e.parameter.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 使用 switch 結構提高可讀性與效能
    switch (action) {
      
      // --- 分支 1：讀取標籤定義 ---
      case 'readTags':
        var tagSheet = ss.getSheetByName("Tags") || ss.insertSheet("Tags");
        var tagRows = tagSheet.getDataRange().getValues();
        var tags = tagRows.slice(1) // 直接切掉標題列
          .filter(row => row[0])    // 過濾空列
          .map(row => ({ name: row[0], definition: row[1] || "" }));
        return createJsonResponse(tags);

      // --- 分支 2：抓取封面 URL ---
      case 'fetchCover':
        return handleFetchCover(e.parameter.url);

      // --- 分支 2b：從 ちるちる 抓書名、作者、封面 ---
      case 'fetchMeta':
        return handleFetchMeta(e.parameter.url);

      // --- 分支 4：讀取待購清單 ---
      case 'readWishlist':
        var wSheet = ss.getSheetByName("Wishlist");
        if (!wSheet) return createJsonResponse([]);
        var wRows = wSheet.getDataRange().getValues();
        if (wRows.length === 0) return createJsonResponse([]);
        // 相容有無標題列：若第一列第一欄是 "timestamp" 則跳過，否則全部讀取
        var wStartRow = (String(wRows[0][0]).toLowerCase() === "timestamp") ? 1 : 0;
        var wData = wRows.slice(wStartRow).filter(function(r) { return r[1]; }).map(function(row) {
          return {
            timestamp:  String(row[0]),
            title:      row[1]  || "",
            jpTitle:    row[2]  || "",
            author:     row[3]  || "",
            reviewer:   row[4]  || "",
            status:     row[5]  || "",
            notes:      row[6]  || "",
            ebookUrl:   row[7]  || "",
            chilUrl:    row[8]  || "",
            coverUrl:   row[9]  || "",
            purchased:  row[10] || ""
          };
        });
        return createJsonResponse(wData);

      // --- 分支 3：讀取書籍資料 (核心優化區) ---
      case 'read':
        var bookSheet = ss.getSheetByName("Books");
        if (!bookSheet) return createJsonResponse({ error: "找不到 Books 工作表" });
        
        var rows = bookSheet.getDataRange().getValues();
        if (rows.length <= 1) return createJsonResponse([]);

        // 使用 map 一次性處理，減少迴圈內部的計算負擔
        var data = rows.slice(1) // 排除標題列
          .filter(row => row[1]) // 確保「書名」欄位不為空
          .map(function(row) {
            return {
              timestamp: row[0] ? String(row[0]) : "",
              title:     row[1] || "",
              jpTitle:   row[2] || "",
              author:    row[3] || "",
              level:     row[4] || "",
              reviewer:  row[5] || "",
              rating:    row[6] || 0,
              tags:      row[7] || "",
              ebookUrl:  row[8] || "",
              chilUrl:   row[9] || "",
              comment:   row[10] || "",
              coverUrl:  row[11] || "",
              twStatus:  row[12] || "",
              jpStatus:  row[13] || ""
            };
          });
        return createJsonResponse(data);

      default:
        return createJsonResponse({ error: "無效的 action 指令" });
    }
  } catch (err) {
    return createJsonResponse({ error: "系統錯誤", detail: err.toString() });
  }
}

/**
 * 處理封面抓取邏輯 (獨立出來讓結構更清晰)
 */
function handleFetchCover(targetUrl) {
  if (!targetUrl) return createJsonResponse({ error: "未提供 URL" });
  
  var coverUrl = "";
  try {
    if (targetUrl.indexOf("/product/") !== -1) {
      var match = targetUrl.match(/\/product\/(\d+)/);
      if (match) coverUrl = "https://taiwan-image.bookwalker.com.tw/product/" + match[1] + "/" + match[1] + "_1.jpg";
    } 
    else if (targetUrl.indexOf("/serial/") !== -1) {
      var match = targetUrl.match(/\/serial\/(\d+)/);
      if (match) coverUrl = "https://taiwan-image.bookwalker.com.tw/series/" + match[1] + ".jpg";
    }
    else if (targetUrl.indexOf("chil-chil.net/goodsDetail/") !== -1) {
      var match = targetUrl.match(/goods_id\/(\d+)\//);
      if (match) {
        var paddedId = match[1].padStart(8, '0');
        coverUrl = "https://img.chil-chil.net/goods_img/XL/" + paddedId + "_XL.jpg";
      }
    }

    if (coverUrl) {
      return createJsonResponse({ coverUrl: coverUrl });
    } else {
      return createJsonResponse({ error: "無法識別此網址格式" });
    }
  } catch (e) {
    return createJsonResponse({ error: "解析封面失敗", detail: e.toString() });
  }
}

/**
 * 從 ちるちる 頁面抓取書名、作者、日文書名、封面
 *
 * 這裡用到 UrlFetchApp.fetch()——GAS 內建的 HTTP 請求工具。
 * 它讓 GAS 代替瀏覽器去抓 ちるちる 的頁面 HTML，
 * 再用 regex（正則表達式）從 HTML 文字裡撈出我們要的資料。
 */
function handleFetchMeta(targetUrl) {
  if (!targetUrl) return createJsonResponse({ error: "未提供 URL" });
  if (targetUrl.indexOf("chil-chil.net/goodsDetail/") === -1) {
    return createJsonResponse({ error: "目前只支援 ちるちる 網址" });
  }

  var result = { coverUrl: "", title: "", jpTitle: "", author: "" };

  try {
    // 1. 封面 URL：從網址裡抓 goods_id，拼出圖片網址（不需要發請求）
    var idMatch = targetUrl.match(/goods_id\/(\d+)\//);
    if (idMatch) {
      result.coverUrl = "https://img.chil-chil.net/goods_img/XL/" +
                        idMatch[1].padStart(8, '0') + "_XL.jpg";
    }

    // 2. 用 UrlFetchApp.fetch() 真正去抓 ちるちる 頁面的 HTML
    //    muteHttpExceptions: true 表示即使回傳 4xx/5xx 也不拋出例外，讓我們自己處理
    var response = UrlFetchApp.fetch(targetUrl, { muteHttpExceptions: true });
    var html = response.getContentText("UTF-8");

    // 3. 日文書名：ちるちる 的 <h1> 就是日文原著書名
    var h1Match = html.match(/<h1[^>]*>([\s\S]*?)<\/h1>/i);
    if (h1Match) result.jpTitle = h1Match[1].replace(/<[^>]+>/g, "").trim();

    // 4. 作者：在 /authorDetail/author_id/XXXXX/ 的連結文字裡
    //    例：<a href=".../authorDetail/author_id/11943/">軟式こんにゃく</a>
    var authorMatch = html.match(/<a[^>]+authorDetail\/author_id\/\d+\/[^>]*>([^<]+)<\/a>/);
    if (authorMatch) result.author = authorMatch[1].trim();

    // 5. 中文書名（title）：ちるちる 是日文網站，沒有中文書名，留空讓使用者手動填

    return createJsonResponse(result);

  } catch (err) {
    return createJsonResponse({ error: "抓取失敗", detail: err.toString() });
  }
}

/**
 * 統一格式化 JSON 回傳
 */
function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var p = e.parameter;
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // --- 待購清單：新增 ---
    if (p.action === "addWishlist") {
      var wSheet = ss.getSheetByName("Wishlist") || ss.insertSheet("Wishlist");
      wSheet.appendRow([
        new Date(), p.title||"", p.jpTitle||"", p.author||"",
        p.reviewer||"", p.status||"", p.notes||"",
        p.ebookUrl||"", p.chilUrl||"", p.coverUrl||"", ""
      ]);
      return createJsonResponse({ status: "Wishlist Add Success" });
    }

    // --- 待購清單：更新 ---
    if (p.action === "updateWishlist") {
      var wSheet = ss.getSheetByName("Wishlist");
      if (!wSheet) return createJsonResponse({ error: "找不到 Wishlist 工作表" });
      var wData = wSheet.getDataRange().getValues();

      // 狀態是書籍屬性，以書名同步更新所有同名列
      if (p.status !== undefined && p.title) {
        for (var i = 0; i < wData.length; i++) {
          if (String(wData[i][1]).trim() === String(p.title).trim()) {
            wSheet.getRange(i+1, 6).setValue(p.status);
          }
        }
      }

      // 備註和已購入是個人屬性，以 timestamp 更新特定列
      if (p.timestamp) {
        for (var j = 0; j < wData.length; j++) {
          if (String(wData[j][0]).trim() === String(p.timestamp).trim()) {
            if (p.purchased !== undefined) wSheet.getRange(j+1, 11).setValue(p.purchased);
            if (p.notes     !== undefined) wSheet.getRange(j+1, 7).setValue(p.notes);
            break;
          }
        }
      }

      return createJsonResponse({ status: "Wishlist Update Success" });
    }

    // --- 待購清單：刪除 ---
    if (p.action === "deleteWishlist") {
      var wSheet = ss.getSheetByName("Wishlist");
      if (!wSheet) return createJsonResponse({ error: "找不到 Wishlist 工作表" });
      var wData = wSheet.getDataRange().getValues();
      for (var i = 0; i < wData.length; i++) {
        if (String(wData[i][0]).trim() === String(p.timestamp).trim()) {
          wSheet.deleteRow(i+1);
          break;
        }
      }
      return createJsonResponse({ status: "Wishlist Delete Success" });
    }

    // --- 分支：新增或更新標籤 ---
    if (p.action === "upsertTag") {
      var tagSheet = ss.getSheetByName("Tags");
      var tagData = tagSheet.getDataRange().getValues();
      var found = false;
      for (var i = 1; i < tagData.length; i++) {
        if (tagData[i][0] === p.tagName) {
          tagSheet.getRange(i + 1, 2).setValue(p.definition);
          found = true;
          break;
        }
      }
      if (!found) tagSheet.appendRow([p.tagName, p.definition]);
      return createJsonResponse({ status: "Tag Success" });
    }

    // --- 分支：刪除標籤 ---
    if (p.action === "deleteTag") {
      var tagSheet = ss.getSheetByName("Tags");
      var tagData = tagSheet.getDataRange().getValues();
      for (var i = 1; i < tagData.length; i++) {
        if (tagData[i][0] === p.tagName) {
          tagSheet.deleteRow(i + 1);
          break;
        }
      }
      return createJsonResponse({ status: "Delete Success" });
    }

    var sheet = ss.getSheetByName("Books");
    if (!sheet) return createJsonResponse({ error: "找不到 Books 工作表" });
    var data = sheet.getDataRange().getValues();

    // --- 分支一：封面自動同步 (updateCoverOnly) ---
    if (p.action === "updateCoverOnly") {
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][1]).trim() === String(p.title).trim()) {
          sheet.getRange(i + 1, 12).setValue(p.coverUrl); // L: 封面
        }
      }
      return createJsonResponse({ status: "Cover Sync Success" });
    }

    // --- 分支二：更新資訊與評語 (update) ---
    if (p.action === "update") {
      var oldTitle = p.oldTitle ? String(p.oldTitle).trim() : "";
      var targetTs = String(p.timestamp).trim();

      for (var i = 1; i < data.length; i++) {
        var sheetTs    = String(data[i][0]).trim();
        var sheetTitle = String(data[i][1]).trim();

        // 1. 同步更新書籍基本資訊（用 !== undefined 確保空字串也能清空欄位）
        if (oldTitle !== "" && sheetTitle === oldTitle) {
          if (p.title    !== undefined) sheet.getRange(i + 1, 2).setValue(p.title);    // B: 書名
          if (p.jpTitle  !== undefined) sheet.getRange(i + 1, 3).setValue(p.jpTitle);  // C: 日文
          if (p.author   !== undefined) sheet.getRange(i + 1, 4).setValue(p.author);   // D: 作者
          if (p.level    !== undefined) sheet.getRange(i + 1, 5).setValue(p.level);    // E: 分級
          if (p.tags     !== undefined) sheet.getRange(i + 1, 8).setValue(p.tags);     // H: 標籤
          if (p.ebookUrl !== undefined) sheet.getRange(i + 1, 9).setValue(p.ebookUrl); // I: 電子書
          if (p.chilUrl  !== undefined) sheet.getRange(i + 1, 10).setValue(p.chilUrl); // J: ちるちる
          if (p.twStatus !== undefined) sheet.getRange(i + 1, 13).setValue(p.twStatus);// M: 台灣狀態
          if (p.jpStatus !== undefined) sheet.getRange(i + 1, 14).setValue(p.jpStatus);// N: 日本狀態
        }

        // 2. 更新個人評語（精確鎖定時間戳）
        if (sheetTs === targetTs) {
          if (p.rating  !== undefined) sheet.getRange(i + 1, 7).setValue(p.rating);   // G: 評分
          if (p.comment !== undefined) sheet.getRange(i + 1, 11).setValue(p.comment); // K: 評語
        }
      }
      return createJsonResponse({ status: "Update Success" });
    }

    // --- 預設：新增一筆評論 ---
    var rowData = [
      new Date(),       // A: 時間戳
      p.title    || "", // B: 書名
      p.jpTitle  || "", // C: 日文
      p.author   || "", // D: 作者
      p.level    || "", // E: 分級
      p.reviewer || "", // F: 評論者
      p.rating   || "", // G: 評分
      p.tags     || "", // H: 標籤
      p.ebookUrl || "", // I: 電子書網址
      p.chilUrl  || "", // J: ちるちる網址
      p.comment  || "", // K: 評語
      p.coverUrl || "", // L: 封面網址
      p.twStatus || "", // M: 台灣狀態
      p.jpStatus || ""  // N: 日本狀態
    ];
    sheet.appendRow(rowData);
    return createJsonResponse({ status: "Insert Success" });

  } catch (err) {
    return createJsonResponse({ error: "系統錯誤", detail: err.toString() });
  }
}

// 輔助函式：統一回傳 JSON
function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
